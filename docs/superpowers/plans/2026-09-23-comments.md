# Comments Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Private, non-printing comment threads anchored to a text range or a block, created from the editor and managed in a new Comments sidebar panel (Beside text + List views).

**Architecture:** Range anchors are a `commentAnchor` mark in the script document (persisted through `script_blocks.content_json`); block anchors and all thread content live in two new DB tables. A Tiptap `CommentsExtension` owns the draft, active/hover state, host-provided thread refs, anchor index, decorations, and block-merge detection. The Comments panel (app-routes) renders inside the editor context, reads editor state through hooks, and persists through an app-core store over the repository — the same layering music uses.

**Tech Stack:** TypeScript, React, Tiptap 3 / ProseMirror, Drizzle + PGlite, TanStack DB (`@tanstack/react-db`), React Aria Components, CSS Modules, Vite+ tests, Moon.

**Spec:** `docs/superpowers/specs/2026-09-23-comments-design.md`

## Global Constraints

- Feature name in UI and code is **Comments** (never "Notes"; `note` is an existing printed block type).
- Range anchor: mark `commentAnchor`, attr `threadId`, `excludes: ''`, `inclusive: false`.
- Block anchor: `script_comment_threads.anchor_block_id`; never a document/block attr.
- `SCRIPT_DOCUMENT_SCHEMA_VERSION` goes 3 → 4 (once, Task 1).
- Anchor add/remove transactions set `addToHistory: false`.
- Paste strips `commentAnchor` marks.
- Comments are in `.stepkg` only; `.stagistic` and PDF output unchanged.
- Old `.stepkg` files without `data/comments.json` must still import (comments = empty).
- Every repo write handler calls `syncDb()` after its transaction.
- Author ids use `LOCAL_COMMENT_AUTHOR_ID = 'local'`; UI never shows authors.
- Status values: `open` | `resolved`. Anchor kinds: `range` | `block`.
- Shortcut: `Mod-Alt-m` (⌘⌥M / Ctrl+Alt+M).
- Empty state copy exactly: `No comments. Select text and press ⌘⌥M.`
- Palette: underline/marker use existing Creamed Corn tokens; never add a green/lime hue. Whole-number pixel values only.
- DB schema change → run `moon run db:db-compile-migrations`. Migrations are hand-written SQL (drizzle-kit generate drifted).
- Use Moon for format, lint, typecheck, tests. Tests import from `vite-plus/test`.
- Run `graphify update .` after code changes.
- Never commit. At each "Commit" step, stage nothing; prepare the suggested commit message and hand it to the user.
- Track progress in repo-root `TODO.md` mirroring this plan's task checkboxes; tick each immediately when done.

```ts
import type {ScriptDocument, ScriptNode} from '../document';

export const COMMENT_ANCHOR_MARK_NAME = 'commentAnchor';
export const COMMENT_THREAD_ID_ATTR = 'threadId';

type InlineMark = NonNullable<ScriptNode['marks']>[number];

const readCommentThreadId = (mark: InlineMark): string | null => {
    if (mark.type !== COMMENT_ANCHOR_MARK_NAME) {
        return null;
    }

    const value = mark.attrs?.[COMMENT_THREAD_ID_ATTR];

    return typeof value === 'string' && value.length > 0 ? value : null;
};

const visit = (node: ScriptNode, ids: Set<string>) => {
    node.marks?.forEach(mark => {
        const threadId = readCommentThreadId(mark);

        if (threadId) {
            ids.add(threadId);
        }
    });
    node.content?.forEach(child => visit(child, ids));
};

export const collectCommentAnchorThreadIds = (document: ScriptDocument): Set<string> => {
    const ids = new Set<string>();

    document.content.forEach(node => visit(node, ids));

    return ids;
};
```

```ts
import {and, asc, eq, type InferInsertModel} from 'drizzle-orm';

import {scriptCommentMessages, scriptCommentThreads} from '../../schema';
import type {DbClient} from '../types';

export type InsertScriptCommentThreadRow = InferInsertModel<typeof scriptCommentThreads>;
export type InsertScriptCommentMessageRow = InferInsertModel<typeof scriptCommentMessages>;

export const listScriptCommentThreads = async (db: DbClient, scriptId: string) => {
    return db.select().from(scriptCommentThreads)
        .where(eq(scriptCommentThreads.scriptId, scriptId))
        .orderBy(asc(scriptCommentThreads.createdAt), asc(scriptCommentThreads.id));
};

export const listScriptCommentMessages = async (db: DbClient, scriptId: string) => {
    return db.select().from(scriptCommentMessages)
        .where(eq(scriptCommentMessages.scriptId, scriptId))
        .orderBy(asc(scriptCommentMessages.threadId), asc(scriptCommentMessages.createdAt), asc(scriptCommentMessages.id));
};

export const listScriptCommentMessagesByThread = async (db: DbClient, payload: {scriptId: string; threadId: string}) => {
```

```ts
import * as dbQueries from '../queries';
import type {ScriptCommentsRepository} from '../scriptRepository';
import {LOCAL_COMMENT_AUTHOR_ID} from '../types';
import type {GetDb, RecordOutbox, SyncDb} from './types';

interface CreateCommentHandlersArgs {
    getDb: GetDb,
    recordOutbox: RecordOutbox,
    syncDb: SyncDb,
}

export const createCommentHandlers = ({getDb, recordOutbox, syncDb}: CreateCommentHandlersArgs): ScriptCommentsRepository => {
    const touch = async (
        tx: Parameters<Parameters<Awaited<ReturnType<GetDb>>['transaction']>[0]>[0],
        scriptId: string,
        entityKey: string,
        opType: string,
        now: number,
        payload: Record<string, unknown>,
    ) => {
        await dbQueries.updateScriptTimestamp(tx, {scriptId, updatedAt: now});
        await recordOutbox({scriptId, entityKey, opType, occurredAt: now, payloadJson: JSON.stringify({scriptId, ...payload})}, tx);
    };

    const listThreads: ScriptCommentsRepository['listThreads'] = async scriptId => dbQueries.listScriptCommentThreads(await getDb(), scriptId);
    const listMessages: ScriptCommentsRepository['listMessages'] = async scriptId => dbQueries.listScriptCommentMessages(await getDb(), scriptId);

    const create: ScriptCommentsRepository['create'] = async (scriptId, input) => {
        const body = input.body.trim();

        if (!body) {
            return null;
        }

        const db = await getDb();
        const now = input.timestamp ?? Date.now();

        await db.transaction(async tx => {
            await dbQueries.insertScriptCommentThread(tx, {
                id: input.id,
                scriptId,
                anchorKind: input.anchorKind,
                anchorBlockId: input.anchorKind === 'block' ? input.anchorBlockId : null,
                quotedText: input.quotedText,
                status: 'open',
                resolvedAt: null,
                resolvedBy: null,
                createdBy: LOCAL_COMMENT_AUTHOR_ID,
                createdAt: now,
                updatedAt: now,
            });
            await dbQueries.insertScriptCommentMessage(tx, {
                id: input.messageId, scriptId, threadId: input.id, authorId: LOCAL_COMMENT_AUTHOR_ID, body, createdAt: now, updatedAt: now, editedAt: null,
            });
            await touch(tx, scriptId, `comment:${input.id}`, 'comment.create', now, {threadId: input.id, messageId: input.messageId});
        });
        await syncDb();

        return dbQueries.getScriptCommentThreadById(db, {scriptId, threadId: input.id});
    };

    const addMessage: ScriptCommentsRepository['addMessage'] = async (scriptId, input) => {
        const body = input.body.trim();

        if (!body) {
            return null;
        }

        const db = await getDb();
        const now = input.timestamp ?? Date.now();

        await db.transaction(async tx => {
            await dbQueries.insertScriptCommentMessage(tx, {
                id: input.id, scriptId, threadId: input.threadId, authorId: LOCAL_COMMENT_AUTHOR_ID, body, createdAt: now, updatedAt: now, editedAt: null,
            });
            await touch(tx, scriptId, `comment:${input.threadId}`, 'comment.message.add', now, {threadId: input.threadId, messageId: input.id});
        });
        await syncDb();

        return dbQueries.getScriptCommentMessageById(db, {scriptId, messageId: input.id});
    };

    const updateMessage: ScriptCommentsRepository['updateMessage'] = async (scriptId, messageId, rawBody) => {
        const body = rawBody.trim();

        if (!body) {
            return null;
        }

        const db = await getDb();
        const now = Date.now();

        await db.transaction(async tx => {
            await dbQueries.updateScriptCommentMessageBody(tx, {scriptId, messageId, body, editedAt: now});
            await touch(tx, scriptId, `comment-message:${messageId}`, 'comment.message.update', now, {messageId});
        });
        await syncDb();

        return dbQueries.getScriptCommentMessageById(db, {scriptId, messageId});
    };

    const deleteMessage: ScriptCommentsRepository['deleteMessage'] = async (scriptId, messageId) => {
        const db = await getDb();
        const now = Date.now();

        await db.transaction(async tx => {
            await dbQueries.deleteScriptCommentMessage(tx, {scriptId, messageId});
            await touch(tx, scriptId, `comment-message:${messageId}`, 'comment.message.delete', now, {messageId});
        });
        await syncDb();
    };

    const setStatus: ScriptCommentsRepository['setStatus'] = async (scriptId, threadId, status) => {
        const db = await getDb();
        const now = Date.now();
        const isResolved = status === 'resolved';

        await db.transaction(async tx => {
            await dbQueries.updateScriptCommentThreadStatus(tx, {
                scriptId, threadId, status, resolvedAt: isResolved ? now : null, resolvedBy: isResolved ? LOCAL_COMMENT_AUTHOR_ID : null, updatedAt: now,
            });
            await touch(tx, scriptId, `comment:${threadId}`, isResolved ? 'comment.resolve' : 'comment.reopen', now, {threadId});
        });
        await syncDb();

        return dbQueries.getScriptCommentThreadById(db, {scriptId, threadId});
    };

    const moveBlockAnchors: ScriptCommentsRepository['moveBlockAnchors'] = async (scriptId, fromBlockId, toBlockId) => {
        const db = await getDb();
        const now = Date.now();
        let moved: string[] = [];

        await db.transaction(async tx => {
            moved = await dbQueries.moveScriptCommentBlockAnchors(tx, {scriptId, fromBlockId, toBlockId, updatedAt: now});

            if (moved.length > 0) {
                await touch(tx, scriptId, `comment-block:${fromBlockId}`, 'comment.anchor.move', now, {fromBlockId, toBlockId, threadIds: moved});
            }
        });

        if (moved.length > 0) {
            await syncDb();
        }
    };

    const deleteThread: ScriptCommentsRepository['delete'] = async (scriptId, threadId) => {
        const db = await getDb();
        const now = Date.now();

        await db.transaction(async tx => {
            await dbQueries.deleteScriptCommentThread(tx, {scriptId, threadId});
            await touch(tx, scriptId, `comment:${threadId}`, 'comment.delete', now, {threadId});
        });
        await syncDb();
    };

    const restore: ScriptCommentsRepository['restore'] = async (scriptId, snapshot) => {
        const db = await getDb();
        const now = Date.now();

        await db.transaction(async tx => {
            await dbQueries.insertScriptCommentThread(tx, {...snapshot.thread, scriptId});
            await dbQueries.bulkInsertScriptCommentMessages(tx, snapshot.messages.map(row => ({...row, scriptId})));
            await touch(tx, scriptId, `comment:${snapshot.thread.id}`, 'comment.restore', now, {threadId: snapshot.thread.id});
        });
        await syncDb();
    };

    return {listThreads, listMessages, create, addMessage, updateMessage, deleteMessage, setStatus, moveBlockAnchors, delete: deleteThread, restore};
};
```

```ts
import type {
    CommentThreadStatus,
    CreateScriptCommentThreadInput,
    ScriptCommentMessage,
    ScriptCommentThread,
    ScriptCommentThreadSnapshot,
    ScriptRepository,
} from '@stagistic/db';

import {createReactiveCollection, createRepositoryStoreRegistry, toDomainCollectionValue} from '../collections';

export type CreateCommentThreadInput = Omit<CreateScriptCommentThreadInput, 'messageId' | 'timestamp'>;

export const createScriptCommentsStore = (repository: ScriptRepository, scriptId: string) => {
    const threads = createReactiveCollection<ScriptCommentThread, string>({
        id: `script-comment-threads:${scriptId}`,
        source: repository.getScriptCommentThreadsSource(scriptId),
        getKey: thread => thread.id,
    });
    const messages = createReactiveCollection<ScriptCommentMessage, string>({
        id: `script-comment-messages:${scriptId}`,
        source: repository.getScriptCommentMessagesSource(scriptId),
        getKey: message => message.id,
    });

    const ready = async () => {
        await Promise.all([threads.collection.preload(), messages.collection.preload()]);
    };

    const readThreadSnapshot = (threadId: string): ScriptCommentThreadSnapshot | null => {
        const thread = threads.collection.get(threadId);

        if (!thread) {
            return null;
        }

        return {
            thread: toDomainCollectionValue(thread),
            messages: [...messages.collection.values()]
                .filter(message => message.threadId === threadId)
                .map(message => toDomainCollectionValue(message))
                .sort((left, right) => left.createdAt - right.createdAt),
        };
    };

    return {
        threadsCollection: threads.collection,
        messagesCollection: messages.collection,
        threadsStatus: threads.status,
        messagesStatus: messages.status,
        ready,
        allocateThreadId: () => repository.allocateScriptCommentThreadId(),
        createThread: (input: CreateCommentThreadInput) => repository.createScriptCommentThread(scriptId, {
            ...input,
            messageId: repository.allocateScriptCommentMessageId(),
        }),
        reply: (threadId: string, body: string) => repository.addScriptCommentMessage(scriptId, {
            id: repository.allocateScriptCommentMessageId(),
            threadId,
            body,
        }),
        editMessage: (messageId: string, body: string) => repository.updateScriptCommentMessage(scriptId, messageId, body),
        deleteMessage: (messageId: string) => repository.deleteScriptCommentMessage(scriptId, messageId),
        setStatus: async (threadId: string, status: CommentThreadStatus) => {
            await repository.setScriptCommentThreadStatus(scriptId, threadId, status);
        },
        deleteThread: async (threadId: string) => {
            const snapshot = readThreadSnapshot(threadId);

            await repository.deleteScriptCommentThread(scriptId, threadId);

            return snapshot;
        },
        restoreThread: (snapshot: ScriptCommentThreadSnapshot) => repository.restoreScriptCommentThread(scriptId, snapshot),
        moveBlockAnchors: (fromBlockId: string, toBlockId: string) => repository.moveScriptCommentBlockAnchors(scriptId, fromBlockId, toBlockId),
    };
};

export type ScriptCommentsStore = ReturnType<typeof createScriptCommentsStore>;
```

```ts
import {COMMENT_ANCHOR_MARK_NAME, COMMENT_THREAD_ID_ATTR} from '@stagistic/script';
import {Mark, mergeAttributes} from '@tiptap/core';
import {Fragment, type Node as ProseMirrorNode, Slice} from '@tiptap/pm/model';
import {Plugin, PluginKey} from '@tiptap/pm/state';

const stripAnchorMarks = (fragment: Fragment): Fragment => {
    const children: ProseMirrorNode[] = [];

    fragment.forEach(child => {
        const marks = child.marks.filter(mark => mark.type.name !== COMMENT_ANCHOR_MARK_NAME);
        const next = child.isText ? child.mark(marks) : child.copy(stripAnchorMarks(child.content)).mark(marks);

        children.push(next);
    });

    return Fragment.fromArray(children);
};

export const CommentAnchorMark = Mark.create({
    name: COMMENT_ANCHOR_MARK_NAME,
    inclusive: false,
    excludes: '',

    addAttributes() {
        return {
            [COMMENT_THREAD_ID_ATTR]: {
                default: '',
                parseHTML: element => element.getAttribute('data-comment-thread-id') ?? '',
                renderHTML: attributes => ({'data-comment-thread-id': String(attributes[COMMENT_THREAD_ID_ATTR] ?? '')}),
            },
        };
    },

    parseHTML() {
        return [{tag: 'span[data-comment-thread-id]'}];
    },

    renderHTML({HTMLAttributes}) {
        return ['span', mergeAttributes(HTMLAttributes), 0];
    },

    addProseMirrorPlugins() {
        return [
            new Plugin({
                key: new PluginKey('comment-anchor-paste'),
                props: {
                    transformPasted: slice => new Slice(stripAnchorMarks(slice.content), slice.openStart, slice.openEnd),
                },
            }),
        ];
    },
```

Semantics:
- `startCommentDraft`: non-empty `TextSelection` → `kind: 'range'`, `from/to` = selection, `blockId` = block containing `from`, `quotedText` = `doc.textBetween(from, to, ' ')`. Empty selection → `kind: 'block'`, active block content range, `quotedText` = block text. Sets `activeThreadId = null`. Calls `onRequestReveal` (from the plugin view `update`, when `draft` goes from null to non-null).
- `commitCommentDraft(threadId)`: range → `tr.addMark(from, to, commentAnchor{threadId})`; block → no doc change. Clears draft, sets `activeThreadId = threadId`. `addToHistory: false`.
- `removeCommentAnchor(threadId)`: records `tombstones[threadId] = {from, to}` from the current anchor, removes marks with that `threadId` across the doc, `addToHistory: false`.
- `restoreCommentAnchor(threadId)`: re-adds the mark at the (mapped) tombstone range if `from < to` and within doc bounds; deletes the tombstone. `addToHistory: false`.
- Tombstones and draft ranges are mapped through every doc-changing transaction; a draft whose range fully collapses (`from >= to` for range) is cleared.
- Decorations:
  - open range anchor → `Decoration.inline(from, to, {class: styles.anchor, 'data-comment-anchor': threadId})`;
  - active or hovered thread → add `styles.anchorActive` (range) or `Decoration.node(blockPos, blockEnd, {class: styles.blockActive})` (block);
  - draft range → `Decoration.inline(..., {class: styles.draft})`; block draft → `Decoration.node(..., {class: styles.blockActive})`.
  - Resolved and unknown-thread anchors get no decoration.
- `handleClick(view, pos)`: collect `threadId`s of open anchors covering `pos`; if any, call `onAnchorClick(ids)`; return `false` (never swallow caret placement).
- Keyboard: `Mod-Alt-m` → `startCommentDraft()`; returns `true`.
- `mergedBlocks` is recomputed per doc-changing transaction via `detectMergedBlocks` **only when the new top-level block count is smaller**, and flushed to `onBlocksMerged` in the plugin view `update` when non-empty. Undo/redo transactions (meta `history$` present) do not report merges.

```ts
import {COMMENT_ANCHOR_MARK_NAME, COMMENT_THREAD_ID_ATTR} from '@stagistic/script';
import type {Node as ProseMirrorNode} from '@tiptap/pm/model';

import type {CommentAnchorLocation, EditorCommentThreadRef} from './types';

const isBlockNode = (node: ProseMirrorNode) => node.isTextblock && typeof node.attrs.id === 'string';

export const buildCommentAnchorIndex = (
    doc: ProseMirrorNode,
    threads: ReadonlyMap<string, EditorCommentThreadRef>,
): Map<string, CommentAnchorLocation> => {
    const index = new Map<string, CommentAnchorLocation>();
    const blockAnchorsByBlockId = new Map<string, string[]>();
    let blockIndex = -1;
    let scene: {id: string, title: string} | null = null;

    threads.forEach(thread => {
        if (thread.anchorKind === 'block' && thread.anchorBlockId) {
            blockAnchorsByBlockId.set(thread.anchorBlockId, [...(blockAnchorsByBlockId.get(thread.anchorBlockId) ?? []), thread.id]);
        }
    });

    doc.descendants((node, pos) => {
        if (!isBlockNode(node)) {
            return true;
        }

        blockIndex += 1;

        const blockId = String(node.attrs.id);

        if (node.attrs.blockType === 'scene' || node.type.name === 'scene') {
            scene = {id: blockId, title: node.textContent};
        }

        const contentFrom = pos + 1;
        const currentScene = scene;
        const base = {blockId, blockIndex, sceneBlockId: currentScene?.id ?? null, sceneTitle: currentScene?.title ?? ''};

        blockAnchorsByBlockId.get(blockId)?.forEach(threadId => {
            if (index.has(threadId)) {
                return;
            }

            index.set(threadId, {threadId, kind: 'block', from: contentFrom, to: contentFrom + node.content.size, ...base});
        });

        node.forEach((child, offset) => {
            child.marks.forEach(mark => {
                if (mark.type.name !== COMMENT_ANCHOR_MARK_NAME) {
                    return;
                }

                const threadId = String(mark.attrs[COMMENT_THREAD_ID_ATTR] ?? '');

                if (!threadId) {
                    return;
                }

                const from = contentFrom + offset;
                const to = from + child.nodeSize;
                const existing = index.get(threadId);

                index.set(threadId, existing && existing.kind === 'range'
                    ? {...existing, from: Math.min(existing.from, from), to: Math.max(existing.to, to)}
                    : {threadId, kind: 'range', from, to, ...base});
            });
        });

        return false;
    });

    return index;
};
```

`detectMergedBlocks.ts`:

```ts
import type {Node as ProseMirrorNode} from '@tiptap/pm/model';
import type {Mapping} from '@tiptap/pm/transform';

import type {CommentBlockMerge} from './types';

const collectBlocks = (doc: ProseMirrorNode) => {
    const blocks: {id: string, pos: number, size: number}[] = [];

    doc.descendants((node, pos) => {
        if (node.isTextblock && typeof node.attrs.id === 'string') {
            blocks.push({id: node.attrs.id, pos, size: node.content.size});

            return false;
        }

        return true;
    });

    return blocks;
};

const findBlockIdAt = (doc: ProseMirrorNode, pos: number): string | null => {
    const $pos = doc.resolve(Math.min(Math.max(pos, 0), doc.content.size));

    for (let depth = $pos.depth; depth > 0; depth -= 1) {
        const node = $pos.node(depth);

        if (node.isTextblock && typeof node.attrs.id === 'string') {
            return node.attrs.id;
        }
    }

    return null;
};

export const detectMergedBlocks = (before: ProseMirrorNode, after: ProseMirrorNode, mapping: Pick<Mapping, 'mapResult'>): CommentBlockMerge[] => {
    const afterIds = new Set(collectBlocks(after).map(block => block.id));
    const merges: CommentBlockMerge[] = [];

    collectBlocks(before).forEach(block => {
        if (afterIds.has(block.id) || block.size === 0) {
            return;
        }

        // The block's last content position survives a join; it is deleted when the whole block goes.
        const result = mapping.mapResult(block.pos + block.size, -1);

        if (result.deletedBefore) {
            return;
        }

        const toBlockId = findBlockIdAt(after, result.pos);

        if (toBlockId && toBlockId !== block.id) {
            merges.push({fromBlockId: block.id, toBlockId});
        }
    });

    return merges;
};
```

```ts
    const grouped = new Map<string, string[]>();

    [...anchors.values()]
        .filter(anchor => threads.get(anchor.threadId)?.status === 'open')

        .forEach(anchor => grouped.set(anchor.blockId, [...(grouped.get(anchor.blockId) ?? []), anchor.threadId]));

    doc.descendants((node, pos) => {
        if (range) {
            return false;
        }

        if (node.isTextblock && node.attrs.id === blockId) {
            range = {pos, end: pos + node.nodeSize};

            return false;
        }

        return true;
    });

    return range as {pos: number, end: number} | null;

const mapDraft = (draft: CommentDraft | null, tr: Transaction): CommentDraft | null => {
    if (!draft) {
        return null;

    const to = tr.mapping.map(draft.to, -1);

    if (draft.kind === 'range' && from >= to) {
        return null;
    }

    return {...draft, from, to};

const readDraft = (state: EditorState): CommentDraft | null => {

    if (!selection.empty) {

    return {kind: 'block', blockId: block.id, from: block.from, to: block.to, quotedText: block.node.textContent};

            restoreCommentAnchor: (threadId: string) => ReturnType;
            setActiveCommentThread: (threadId: string | null) => ReturnType;
            setHoveredCommentThread: (threadId: string | null) => ReturnType;
        };
    }
}

export interface CommentsExtensionOptions {
    callbacksRef: {current: CommentsExtensionCallbacks},
}

const withMeta = (tr: Transaction, meta: CommentsMeta) => tr.setMeta(commentsPluginKey, meta).setMeta('addToHistory', false);

export const CommentsExtension = Extension.create<CommentsExtensionOptions>({
    name: 'comments',

    addOptions() {
        return {callbacksRef: {current: {}}};
    },

    addKeyboardShortcuts() {
        return {
            'Mod-Alt-m': () => this.editor.commands.startCommentDraft(),
        };
    },

    addCommands() {
        return {
            setCommentThreads: threads => ({tr, dispatch}) => {
                dispatch?.(withMeta(tr, {type: 'threads', threads}));

                return true;
            },
            startCommentDraft: () => ({tr, state, dispatch}) => {
                const draft = readDraft(state);

                if (!draft) {
                    return false;
                }

                dispatch?.(withMeta(tr, {type: 'draft', draft}));

                return true;
            },
            cancelCommentDraft: () => ({tr, dispatch}) => {
                dispatch?.(withMeta(tr, {type: 'draft', draft: null}));

                return true;
            },
            commitCommentDraft: threadId => ({tr, state, dispatch}) => {
                const draft = getCommentsState(state).draft;

                if (!draft) {
                    return false;
                }

                if (dispatch) {
                    if (draft.kind === 'range') {
                        tr.addMark(draft.from, draft.to, state.schema.marks[COMMENT_ANCHOR_MARK_NAME].create({[COMMENT_THREAD_ID_ATTR]: threadId}));
                    }

                    withMeta(tr, {type: 'draft', draft: null});
                    tr.setMeta('commentsActivate', threadId);
                    dispatch(tr);
                }

                return true;
            },
            removeCommentAnchor: threadId => ({tr, state, dispatch}) => {
                const anchor = getCommentsState(state).anchors.get(threadId);

                if (dispatch) {
                    const markType = state.schema.marks[COMMENT_ANCHOR_MARK_NAME];

                    state.doc.descendants((node, pos) => {
                        node.marks
                            .filter(mark => mark.type === markType && mark.attrs[COMMENT_THREAD_ID_ATTR] === threadId)
                            .forEach(mark => tr.removeMark(pos, pos + node.nodeSize, mark));
                    });
                    dispatch(withMeta(tr, {type: 'tombstone', threadId, tombstone: anchor?.kind === 'range' ? {from: anchor.from, to: anchor.to} : null}));
                }

                return true;
            },
            restoreCommentAnchor: threadId => ({tr, state, dispatch}) => {
                const tombstone = getCommentsState(state).tombstones.get(threadId);

                if (dispatch) {
                    if (tombstone && tombstone.from < tombstone.to && tombstone.to <= state.doc.content.size) {
                        tr.addMark(tombstone.from, tombstone.to, state.schema.marks[COMMENT_ANCHOR_MARK_NAME].create({[COMMENT_THREAD_ID_ATTR]: threadId}));
                    }

                    dispatch(withMeta(tr, {type: 'tombstone', threadId, tombstone: null}));
                }

                return true;
            },
            setActiveCommentThread: threadId => ({tr, dispatch}) => {
                dispatch?.(withMeta(tr, {type: 'active', threadId}));

                return true;
            },
            setHoveredCommentThread: threadId => ({tr, dispatch}) => {
                dispatch?.(withMeta(tr, {type: 'hovered', threadId}));

                return true;
            },
        };
    },

    addProseMirrorPlugins() {
        const {callbacksRef} = this.options;

        return [
            new Plugin<CommentsPluginState>({
                key: commentsPluginKey,
                state: {
                    init: (_config, state) => finalize(state.doc, {
                        threads: new Map(), draft: null, activeThreadId: null, hoveredThreadId: null, tombstones: new Map(), mergedBlocks: [],
                    }),
                    apply: (tr, previous, oldState) => {
                        const meta = tr.getMeta(commentsPluginKey) as CommentsMeta | undefined;
                        const activate = tr.getMeta('commentsActivate') as string | undefined;

                        if (!tr.docChanged && !meta && !activate) {
                            return previous.mergedBlocks.length === 0 ? previous : {...previous, mergedBlocks: []};
                        }

                        let threads = previous.threads;
                        let draft = tr.docChanged ? mapDraft(previous.draft, tr) : previous.draft;
                        let activeThreadId = previous.activeThreadId;
                        let hoveredThreadId = previous.hoveredThreadId;
                        let tombstones: Map<string, CommentTombstone> = new Map(previous.tombstones);

                        if (tr.docChanged) {
                            tombstones = new Map([...tombstones].map(([id, t]) => [id, {from: tr.mapping.map(t.from, 1), to: tr.mapping.map(t.to, -1)}]));
                        }

                        if (meta?.type === 'threads') threads = new Map(meta.threads.map(thread => [thread.id, thread]));
                        if (meta?.type === 'draft') {
                            draft = meta.draft;
                            if (meta.draft) activeThreadId = null;
                        }
                        if (meta?.type === 'active') activeThreadId = meta.threadId;
                        if (meta?.type === 'hovered') hoveredThreadId = meta.threadId;
                        if (meta?.type === 'tombstone') {
                            if (meta.tombstone) tombstones.set(meta.threadId, meta.tombstone);
                            else tombstones.delete(meta.threadId);
                        }
                        if (activate) activeThreadId = activate;

                        const isHistory = Boolean(tr.getMeta('history$'));
                        const mergedBlocks = tr.docChanged && !isHistory && tr.doc.childCount < oldState.doc.childCount
                            ? detectMergedBlocks(oldState.doc, tr.doc, tr.mapping)
                            : [];

                        return finalize(tr.doc, {threads, draft, activeThreadId, hoveredThreadId, tombstones, mergedBlocks});
                    },
                },
                props: {
                    decorations: state => commentsPluginKey.getState(state)?.decorations,
                    handleClick: (view, pos) => {
                        const pluginState = commentsPluginKey.getState(view.state);
                        const ids = [...(pluginState?.anchors.values() ?? [])]
                            .filter(anchor => anchor.kind === 'range' && anchor.from <= pos && pos <= anchor.to)
                            .filter(anchor => pluginState?.threads.get(anchor.threadId)?.status === 'open')
                            .map(anchor => anchor.threadId);

                        if (ids.length > 0) {
                            callbacksRef.current.onAnchorClick?.(ids);
                        }

                        return false;
                    },
                },
                view: () => ({
                    update: (view, previousState) => {
                        const next = commentsPluginKey.getState(view.state);
                        const previous = commentsPluginKey.getState(previousState);

                        if (next?.draft && !previous?.draft) {
                            callbacksRef.current.onRequestReveal?.();
                        }

                        if (next && next.mergedBlocks.length > 0 && next.mergedBlocks !== previous?.mergedBlocks) {
                            callbacksRef.current.onBlocksMerged?.(next.mergedBlocks);
                        }
                    },
                }),
            }),
        ];
    },
});
```

`comments/useCommentAnchorTops.ts`:

```ts
import {useEffect, useState} from 'react';

import {useEditorInstance} from '../context';
import {commentsPluginKey} from '../tiptap/extensions/comments';

const SCROLL_CONTAINER_SELECTOR = '[data-editor-scroll-container="true"]';

const shallowEqual = (left: ReadonlyMap<string, number>, right: ReadonlyMap<string, number>) =>
    left.size === right.size && [...left].every(([key, value]) => right.get(key) === value);

export const useCommentAnchorTops = (threadIds: readonly string[]): ReadonlyMap<string, number> => {
    const editor = useEditorInstance();
    const [tops, setTops] = useState<ReadonlyMap<string, number>>(new Map());
    const key = threadIds.join('|');

    useEffect(() => {
        if (!editor) {
            return undefined;
        }

        const scrollContainer = editor.view.dom.closest(SCROLL_CONTAINER_SELECTOR);
        let frame = 0;

        const measure = () => {
            frame = 0;

            const anchors = commentsPluginKey.getState(editor.state)?.anchors;
            const next = new Map<string, number>();

            threadIds.forEach(threadId => {
                const anchor = anchors?.get(threadId);

                if (!anchor) {
                    return;
                }

                const blockDom = editor.view.nodeDOM(anchor.from - 1);
                const rect = blockDom instanceof HTMLElement ? blockDom.getBoundingClientRect() : null;

                if (!rect || rect.height === 0) {
                    return;
                }

                next.set(threadId, anchor.kind === 'range' ? Math.round(editor.view.coordsAtPos(anchor.from).top) : Math.round(rect.top));
            });
            setTops(previous => (shallowEqual(previous, next) ? previous : next));
        };
        const schedule = () => {
            if (!frame) {
                frame = requestAnimationFrame(measure);
            }
        };

        measure();
        editor.on('transaction', schedule);
        scrollContainer?.addEventListener('scroll', schedule, {passive: true});
        window.addEventListener('resize', schedule);

        return () => {
            cancelAnimationFrame(frame);
            editor.off('transaction', schedule);
            scrollContainer?.removeEventListener('scroll', schedule);
            window.removeEventListener('resize', schedule);
        };
        // eslint-disable-next-line react-hooks/exhaustive-deps
    }, [editor, key]);

    return tops;
};
```

```tsx
import type {Editor as TiptapEditor} from '@tiptap/react';
import {type RefObject, useEffect, useState} from 'react';

import {commentsPluginKey} from '../tiptap/extensions/comments';

import styles from './SelectionToolbarOverlay.module.css';

const TOOLBAR_GAP_PX = 8;

interface Position {top: number, left: number}

const readPosition = (editor: TiptapEditor, canvas: HTMLElement): Position | null => {
    const {selection} = editor.state;

    if (!editor.isFocused || selection.empty || commentsPluginKey.getState(editor.state)?.draft) {
        return null;
    }

    const start = editor.view.coordsAtPos(selection.from);
    const end = editor.view.coordsAtPos(selection.to);
    const canvasRect = canvas.getBoundingClientRect();

    return {
        top: Math.round(start.top - canvasRect.top + canvas.scrollTop - TOOLBAR_GAP_PX),
        left: Math.round((Math.min(start.left, end.left) + Math.max(start.right, end.right)) / 2 - canvasRect.left),
    };
};

export const SelectionToolbarOverlay = ({editor, canvasRef}: {editor: TiptapEditor | null, canvasRef: RefObject<HTMLElement | null>}) => {
    const [position, setPosition] = useState<Position | null>(null);
    const [isPointerDown, setIsPointerDown] = useState(false);

    useEffect(() => {
        const canvas = canvasRef.current;

        if (!editor || !canvas) {
            return undefined;
        }

        const update = () => setPosition(readPosition(editor, canvas));
        const handleMouseDown = () => setIsPointerDown(true);
        const handleMouseUp = () => {
            setIsPointerDown(false);
            update();
        };

        editor.on('selectionUpdate', update);
        editor.on('transaction', update);
        editor.on('focus', update);
        editor.on('blur', update);
        editor.view.dom.addEventListener('mousedown', handleMouseDown);
        window.addEventListener('mouseup', handleMouseUp);

        return () => {
            editor.off('selectionUpdate', update);
            editor.off('transaction', update);
            editor.off('focus', update);
            editor.off('blur', update);
            editor.view.dom.removeEventListener('mousedown', handleMouseDown);
            window.removeEventListener('mouseup', handleMouseUp);
        };
    }, [canvasRef, editor]);

    if (!editor || !position || isPointerDown) {
        return null;
    }

    const keepSelection = (event: {preventDefault: () => void}) => event.preventDefault();
    const copy = () => {
        const {from, to} = editor.state.selection;

        void navigator.clipboard.writeText(editor.state.doc.textBetween(from, to, '\n'));
    };

    return (
        <div
            className={styles.toolbar}
            role="toolbar"
            aria-label="Selection actions"
            style={{top: position.top, left: position.left}}
        >
            <button type="button" className={styles.button} onMouseDown={keepSelection} onClick={() => editor.commands.startCommentDraft()}>
                Comment
            </button>
            <button type="button" className={styles.button} onMouseDown={keepSelection} onClick={copy}>
                Copy
            </button>
        </div>
    );
};
```

```tsx
import type {Editor as TiptapEditor} from '@tiptap/react';
import {type RefObject, useEffect, useState} from 'react';

import {commentsPluginKey} from '../tiptap/extensions/comments';
import {findScriptBlockByIdFromState} from '../tiptap/scriptCore';

import styles from './CommentMarkersOverlay.module.css';

interface Marker {blockId: string, threadIds: readonly string[], top: number}

const readMarkers = (editor: TiptapEditor, canvas: HTMLElement): Marker[] => {
    const grouped = commentsPluginKey.getState(editor.state)?.openThreadIdsByBlockId;
    const canvasRect = canvas.getBoundingClientRect();
    const markers: Marker[] = [];

    grouped?.forEach((threadIds, blockId) => {
        const block = findScriptBlockByIdFromState(editor.state, blockId);
        const dom = block ? editor.view.nodeDOM(block.pos) : null;
        const rect = dom instanceof HTMLElement ? dom.getBoundingClientRect() : null;

        if (rect && rect.height > 0) {
            markers.push({blockId, threadIds, top: Math.round(rect.top - canvasRect.top + canvas.scrollTop)});
        }
    });

    return markers;
};

export const CommentMarkersOverlay = ({editor, canvasRef, onRequestReveal}: {
    editor: TiptapEditor | null,
    canvasRef: RefObject<HTMLElement | null>,
    onRequestReveal?: () => void,
}) => {
    const [markers, setMarkers] = useState<Marker[]>([]);

    useEffect(() => {
        const canvas = canvasRef.current;

        if (!editor || !canvas) {
            return undefined;
        }

        let frame = 0;
        const schedule = () => {
            if (!frame) {
                frame = requestAnimationFrame(() => {
                    frame = 0;
                    setMarkers(readMarkers(editor, canvas));
                });
            }
        };
        const observer = new ResizeObserver(schedule);

        observer.observe(editor.view.dom);
        editor.on('transaction', schedule);
        schedule();

        return () => {
            cancelAnimationFrame(frame);
            observer.disconnect();
            editor.off('transaction', schedule);
        };
    }, [canvasRef, editor]);

    if (!editor) {
        return null;
    }

    return (
        <>
            {markers.map(marker => (
                <button
                    key={marker.blockId}
                    type="button"
                    className={styles.marker}
                    style={{top: marker.top}}
                    data-comment-marker-block-id={marker.blockId}
                    aria-label={marker.threadIds.length === 1 ? '1 comment' : `${marker.threadIds.length} comments`}
                    onMouseDown={event => event.preventDefault()}
                    onClick={() => {
                        editor.commands.setActiveCommentThread(marker.threadIds[0] ?? null);
                        onRequestReveal?.();
                    }}
                >
                    {marker.threadIds.length > 1 ? marker.threadIds.length : ''}
                </button>
            ))}
        </>
    );
};
```

`layoutBesideCards.ts`:

```ts
export interface BesideCardInput {threadId: string, blockId: string, anchorTop: number, height: number}
export interface BesideCardEntry {key: string, blockId: string, threadIds: readonly string[], top: number, height: number, collapsed: boolean}

export const COLLAPSE_THRESHOLD = 3;

interface LayoutArgs {
    cards: readonly BesideCardInput[],
    activeThreadId: string | null,
    expandedBlockId: string | null,
    collapsedHeight: number,
    gap: number,
}

export const layoutBesideCards = ({cards, activeThreadId, expandedBlockId, collapsedHeight, gap}: LayoutArgs): BesideCardEntry[] => {
    const groups = new Map<string, BesideCardInput[]>();

    cards.forEach(card => groups.set(card.blockId, [...(groups.get(card.blockId) ?? []), card]));

    const entries: (Omit<BesideCardEntry, 'top'> & {desired: number, order: number})[] = [];

    groups.forEach((group, blockId) => {
        const shouldCollapse = group.length >= COLLAPSE_THRESHOLD
            && blockId !== expandedBlockId
            && !group.some(card => card.threadId === activeThreadId);

        if (shouldCollapse) {
            entries.push({key: `group:${blockId}`, blockId, threadIds: group.map(card => card.threadId), height: collapsedHeight, collapsed: true, desired: group[0].anchorTop, order: entries.length});

            return;
        }

        group.forEach(card => entries.push({key: card.threadId, blockId, threadIds: [card.threadId], height: card.height, collapsed: false, desired: card.anchorTop, order: entries.length}));
    });

    entries.sort((left, right) => left.desired - right.desired || left.order - right.order);

    const tops = entries.map(entry => entry.desired);
    const activeIndex = entries.findIndex(entry => activeThreadId !== null && entry.threadIds.includes(activeThreadId));
    const start = activeIndex === -1 ? 0 : activeIndex;

    for (let index = start + 1; index < entries.length; index += 1) {
        tops[index] = Math.max(entries[index].desired, tops[index - 1] + entries[index - 1].height + gap);
    }

    if (activeIndex === -1 && entries.length > 0) {
        tops[0] = entries[0].desired;
    }

    for (let index = start - 1; index >= 0; index -= 1) {
        tops[index] = Math.min(entries[index].desired, tops[index + 1] - gap - entries[index].height);
    }

    return entries.map(({desired: _desired, order: _order, ...entry}, index) => ({...entry, top: tops[index]}));
};
```
